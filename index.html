            <button class="btn" onclick="startChoice('all')">全部随机</button>

            <button class="btn" onclick="startSpell('all')">全部随机</button>

            <button class="btn" onclick="showSpellChapters()">按章节学习</button>

            <button class="btn back-home" onclick="showHome()">🏠 返回</button>

        </div>

            <div class="img-box"><img id="game-img" src=""></div>

            <div id="options"></div>

            <div id="letters" class="letter-buttons"></div>

            <button class="btn" id="submit-btn" style="display:none" onclick="checkSpell()">✅ 提交</button>

            <button class="btn" onclick="nextQuestion()">🔄 下一题</button>

            <button class="btn back-home" onclick="showHome()">🏠 返回</button>

    <script>

        // ================== 词库 ==================

        const WORD_BANKS = {

            "My body": ["face","eye","mouth","nose","ear","head","hair","leg","arm","hand","feet","foot"],

            "At the zoo": ["frog","mouse","sheep","tiger","duck","monkey","bird","cow","chicken","horse","lizard","goat","dog","hippo","giraffe","snake","crocodile","elephant","spider","zebra"],

            "At the clothes shop": ["glasses","hat","bag","clothes","handbag","jacket","sock","watch","trousers","shirt","dress","shoe","angry","jeans","T-shirt"],

            "At friend's birthday": ["alien","monster","balloon","baby","children","boy","girl","women","woman","men","man","people","robot","dad","mum","grandma","grandpa","sister","brother","cousin","family"],

            "My favourite food": ["fruit","rice","coconut","bread","onion","tomato","water","milk","dinner","fish","breakfast","watermelon","pineapple","cake","lunch","drink","candy","ice cream","chocolate","carrot","apple","banana","pear","peach","orange","cucumber","hamburger","sausage","lime","juice"],

            "At home": ["kitchen","bedroom","bathroom","house","flat","hall","cat","garden","mirror","bath","cupboard","clock","radio","box","bed","bookcase","painting","mat","television","lamp","armchair","sofa","window","door","chair"],

            "At school": ["draw","ruler","pencil","pen","eraser","page","part","football","classroom","basketball","school","music","tick","cross","open","close","computer","mouse","hockey"],

            "At the beach": ["baseball","walk","paint","fishing","piano","camera","jump","sand","tennis","swim","sun","shell","kite","guitar"],

            "My street": ["run","train","bookshop","shop","tree","park","motorbike","lorry","truck","bus","helicopter","bike","car","plane","stop"]

        };


        // 拼写只保留单个单词

        const SPELL_BANKS = {};

        for (let ch in WORD_BANKS) {

            SPELL_BANKS[ch] = WORD_BANKS[ch].filter(w => w.trim().split(" ").length === 1);

        }


        let allWords = [].concat(...Object.values(WORD_BANKS));

        let allSpellWords = [].concat(...Object.values(SPELL_BANKS));

        let score = 0;

        let currentWord = "";

        let userInput = [];

        let errorWords = JSON.parse(localStorage.getItem("errors") || "[]");

        let gameMode = "";

        let currentChapter = "";

        let isChoiceGame = true;


        // ================== 工具 ==================

        function showHome() {

            document.querySelectorAll("#app > div").forEach(s => s.style.display = "none");

            document.getElementById("screen-home").style.display = "block";

            document.getElementById("score").innerText = "分数：" + score;

        }


        function showChoiceMenu() {

            document.querySelectorAll("#app > div").forEach(s => s.style.display = "none");

            document.getElementById("screen-choice-menu").style.display = "block";

        }


        function showSpellMenu() {

            document.querySelectorAll("#app > div").forEach(s => s.style.display = "none");

            document.getElementById("screen-spell-menu").style.display = "block";

        }


        function getImageSrc(word) {

            if (word === "chicken") return "images/chicken.png";

            if (word === "mouse") return "images/mouse.png";

            return "images/" + word + ".jpg";

        }


        function saveErrors() {

            localStorage.setItem("errors", JSON.stringify(errorWords));

        }


        // ================== 章节选择 ==================

        function showChoiceChapters() {

            renderChapters("choice");

        }


        function showSpellChapters() {

            renderChapters("spell");

        }


        function renderChapters(type) {

            let list = document.getElementById("chapter-list");

            list.innerHTML = "";

            for (let ch in WORD_BANKS) {

                let btn = document.createElement("button");

                btn.className = "btn";

                btn.innerText = ch;

                btn.onclick = () => {

                    if (type === "choice") startChoice("chapter", ch);

                    else startSpell("chapter", ch);

                };

                list.appendChild(btn);

            }

            document.querySelectorAll("#app > div").forEach(s => s.style.display = "none");

            document.getElementById("screen-chapters").style.display = "block";

        }


        // ================== 看图选词 ==================

        function startChoice(mode, chapter = null) {

            gameMode = mode;

            currentChapter = chapter;

            isChoiceGame = true;

            nextQuestion();

        }


        function startErrorReview() {

            if (errorWords.length === 0) {

                alert("错题本为空！");

                return;

            }

            gameMode = "error";

            isChoiceGame = true;

            nextQuestion();

        }


        function nextQuestion() {

            userInput = [];

            let pool = [];


            if (isChoiceGame) {

                if (gameMode === "all") pool = allWords;

                else if (gameMode === "chapter") pool = WORD_BANKS[currentChapter];

                else if (gameMode === "error") pool = errorWords;

            } else {

                if (gameMode === "all") pool = allSpellWords;

                else if (gameMode === "chapter") pool = SPELL_BANKS[currentChapter] || [];

            }


            if (pool.length === 0) {

                alert("暂无题目");

                showHome();

                return;

            }


            currentWord = pool[Math.floor(Math.random() * pool.length)];

            document.getElementById("game-img").src = getImageSrc(currentWord);

            document.querySelectorAll("#app > div").forEach(s => s.style.display = "none");

            document.getElementById("screen-game").style.display = "block";


            if (isChoiceGame) {

                renderChoice(pool);

            } else {

                renderSpell();

            }

        }


        function renderChoice(pool) {

            document.getElementById("word-display").style.display = "none";

            document.getElementById("letters").style.display = "none";

            document.getElementById("submit-btn").style.display = "none";


            let opts = [currentWord];

            while (opts.length < 4) {

                let w = pool[Math.floor(Math.random() * pool.length)];

                if (!opts.includes(w)) opts.push(w);

            }

            opts.sort(() => Math.random() - 0.5);


            let box = document.getElementById("options");

            box.innerHTML = "";

            opts.forEach(o => {

                let b = document.createElement("button");

                b.className = "btn";

                b.innerText = o;

                b.onclick = () => checkAnswer(o);

                box.appendChild(b);

            });

        }


        function checkAnswer(sel) {

            if (sel === currentWord) {

                alert("✅ 正确！");

                score++;

                let i = errorWords.indexOf(currentWord);

                if (i > -1) errorWords.splice(i, 1);

                saveErrors();

            } else {

                alert("❌ 错误，正确：" + currentWord);

                if (!errorWords.includes(currentWord)) {

                    errorWords.push(currentWord);

                    saveErrors();

                }

            }

            nextQuestion();

        }


        // ================== 看图拼词 ==================

        function startSpell(mode, chapter = null) {

            gameMode = mode;

            currentChapter = chapter;

            isChoiceGame = false;

            nextQuestion();

        }


        function renderSpell() {

            document.getElementById("options").innerHTML = "";

            document.getElementById("word-display").style.display = "block";

            document.getElementById("letters").style.display = "flex";

            document.getElementById("submit-btn").style.display = "block";


            userInput = [];

            let chars = currentWord.split("");

            let shuffled = [...chars];

            shuffled.sort(() => Math.random() - 0.5);


            updateWordDisplay();

            let letterBox = document.getElementById("letters");

            letterBox.innerHTML = "";


            shuffled.forEach(c => {

                let btn = document.createElement("button");

                btn.className = "letter-btn";

                btn.innerText = c;

                btn.onclick = () => {

                    if (userInput.length < currentWord.length) {

                        userInput.push(c);

                        updateWordDisplay();

                    }

                };

                letterBox.appendChild(btn);

            });

        }


        function updateWordDisplay() {

            let display = document.getElementById("word-display");

            let txt = [];

            for (let i = 0; i < currentWord.length; i++) {

                txt.push(userInput[i] || "_");

            }

            display.innerText = txt.join("  ");

        }


        function checkSpell() {

            if (userInput.join("") === currentWord) {

                alert("✅ 拼写正确！");

                score++;

                nextQuestion();

            } else {

                alert("❌ 错误，正确：" + currentWord);

            }

        }


        // 初始化

        showHome();

    </script>
